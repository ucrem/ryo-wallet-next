import fs from 'node:fs/promises'
import {createHash} from 'node:crypto'
const manifest=JSON.parse(await fs.readFile('src-tauri/runtime-manifest.json'))
const reports=[]
for(const [name,expected] of [['ryod.exe',manifest.platforms['x86_64-pc-windows-msvc'].daemonSha256],['ryo-wallet-rpc.exe',manifest.platforms['x86_64-pc-windows-msvc'].binarySha256]]){
 const b=await fs.readFile('src-tauri/.dev-runtime/'+name),hash=createHash('sha256').update(b).digest('hex')
 if(hash!==expected)throw Error('Unreviewed runtime')
 const pe=b.readUInt32LE(0x3c),opt=pe+24,sections=b.readUInt16LE(pe+6),optionalSize=b.readUInt16LE(pe+20)
 if(b.toString('ascii',pe,pe+4)!=='PE\0\0'||b.readUInt16LE(opt)!==0x20b)throw Error('Unexpected PE format')
 const table=opt+optionalSize
 function offset(rva){for(let i=0;i<sections;i++){const p=table+i*40,start=b.readUInt32LE(p+12),size=Math.max(b.readUInt32LE(p+8),b.readUInt32LE(p+16));if(rva>=start&&rva<start+size){const o=b.readUInt32LE(p+20)+rva-start;if(o<0||o>=b.length)throw Error('Invalid RVA');return o}}throw Error('Unknown RVA')}
 const imports=[],importRva=b.readUInt32LE(opt+112+8)
 for(let p=offset(importRva),n=0;n<256;n++,p+=20){const nameRva=b.readUInt32LE(p+12);if(!nameRva)break;const s=offset(nameRva),end=b.indexOf(0,s);if(end<0||end-s>256)throw Error('Invalid import name');imports.push(b.toString('ascii',s,end))}
 const markers=[...new Set((b.toString('latin1').match(/[ -~]{12,160}/g)||[]).filter(x=>/OpenSSL [0-9]|LMDB [0-9]|libminiupnpc/.test(x)))].slice(0,20)
 reports.push({name,sha256:hash,pe_imports:imports,version_strings:markers,static_link_composition_proven:false})
}
await fs.writeFile('target/issue-closure-2026-10-06/windows-runtime-imports.json',JSON.stringify(reports,null,2)+'\n')
console.log(JSON.stringify(reports))
